cmake_minimum_required(VERSION 3.24)
project(r3_context LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LLAMA "${ROOT}/third_party/llama.cpp-integrated")
find_package(CUDAToolkit REQUIRED)
link_directories("${CUDAToolkit_LIBRARY_DIR}")
add_executable(r3_capture capture.cpp)
target_include_directories(r3_capture PRIVATE "${LLAMA}/include" "${LLAMA}/ggml/include" "${LLAMA}/vendor")
target_link_libraries(r3_capture PRIVATE "${ROOT}/build-r2-final-20261107/llama/src/llama.lib" "${ROOT}/build-r2-final-20261107/llama/ggml/src/ggml-base.lib")
target_compile_definitions(r3_capture PRIVATE LLAMA_SHARED GGML_SHARED)
target_compile_options(r3_capture PRIVATE /utf-8 /EHsc)
add_executable(r3_context_replay replay.cpp "${ROOT}/experiments/readonly_r3/kernels.cu")
target_include_directories(r3_context_replay PRIVATE "${ROOT}/include" "${ROOT}/third_party/HBF-EventLab/include" "${LLAMA}/vendor" "${ROOT}/experiments/readonly_r3")
target_link_libraries(r3_context_replay PRIVATE "${ROOT}/build-r3-current-20261107/r3_runtime.lib" "${ROOT}/build-r3-current-20261107/eventlab/hbf_eventlab.lib" CUDA::cudart CUDA::cuda_driver)
target_compile_options(r3_context_replay PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
