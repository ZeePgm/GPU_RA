cmake_minimum_required(VERSION 3.24)
project(readonly_r2 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
get_filename_component(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(R1_BUILD "${LAB_ROOT}/build-r1-20261107" CACHE PATH "Frozen R1 runtime")
find_package(CUDAToolkit REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/norm.ptx"
 COMMAND "${CMAKE_CUDA_COMPILER}" --ptx -std=c++17 -arch=compute_75 "${LAB_ROOT}/adapters/llama_cpp/integrated_weight_mul.cu" -o "${CMAKE_CURRENT_BINARY_DIR}/norm.ptx"
 DEPENDS "${LAB_ROOT}/adapters/llama_cpp/integrated_weight_mul.cu" VERBATIM)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/down.ptx"
 COMMAND "${CMAKE_CUDA_COMPILER}" --ptx -std=c++17 -arch=compute_75 "-I${LAB_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/down_probe.cu" -o "${CMAKE_CURRENT_BINARY_DIR}/down.ptx"
 DEPENDS down_probe.cu "${LAB_ROOT}/adapters/llama_cpp/integrated_down_scalar.cu" VERBATIM)
add_executable(readonly_r2 runner.cpp "${LAB_ROOT}/src/ptx.cpp" "${LAB_ROOT}/adapters/llama_cpp/integrated_norm_explicit.cu" norm_timed.cu "${CMAKE_CURRENT_BINARY_DIR}/norm.ptx" "${CMAKE_CURRENT_BINARY_DIR}/down.ptx")
target_include_directories(readonly_r2 PRIVATE "${LAB_ROOT}/include" "${LAB_ROOT}/third_party/HBF-EventLab/include" "${LAB_ROOT}/third_party/llama.cpp-integrated/vendor")
target_link_libraries(readonly_r2 PRIVATE "${R1_BUILD}/realgpu_runtime.lib" "${R1_BUILD}/eventlab/hbf_eventlab.lib" CUDA::cudart CUDA::cuda_driver)
if(MSVC)
 target_compile_options(readonly_r2 PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
endif()

add_executable(readonly_r2_down down_runner.cpp down_explicit.cu down_timed.cu "${LAB_ROOT}/src/ptx.cpp" "${CMAKE_CURRENT_BINARY_DIR}/down.ptx")
target_include_directories(readonly_r2_down PRIVATE "${LAB_ROOT}/include" "${LAB_ROOT}/third_party/HBF-EventLab/include" "${LAB_ROOT}/third_party/llama.cpp-integrated/vendor")
target_link_libraries(readonly_r2_down PRIVATE "${R1_BUILD}/realgpu_runtime.lib" "${R1_BUILD}/eventlab/hbf_eventlab.lib" CUDA::cudart CUDA::cuda_driver)
target_compile_options(readonly_r2_down PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
enable_testing()
foreach(contract ptx_contract async_ptx_contract)
 add_executable(r2_${contract} "${LAB_ROOT}/tests/${contract}.cpp" "${LAB_ROOT}/src/ptx.cpp")
 target_include_directories(r2_${contract} PRIVATE "${LAB_ROOT}/include")
 target_link_libraries(r2_${contract} PRIVATE "${R1_BUILD}/realgpu_runtime.lib")
endforeach()
foreach(case hash coverage predicate reject_control reject_memory reject_module syntax multiple limits arity binary_comments access_manifest large)
 add_test(NAME ptx_${case} COMMAND r2_ptx_contract ${case})
endforeach()
foreach(case control labels predicate reject)
 add_test(NAME async_ptx_${case} COMMAND r2_async_ptx_contract ${case})
endforeach()
add_executable(r2_down_contract down_contract.cpp "${LAB_ROOT}/src/ptx.cpp")
target_include_directories(r2_down_contract PRIVATE "${LAB_ROOT}/include" "${LAB_ROOT}/tests")
add_test(NAME down_extension COMMAND r2_down_contract "${CMAKE_CURRENT_BINARY_DIR}/down.ptx")
