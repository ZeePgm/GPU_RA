cmake_minimum_required(VERSION 3.24)
project(readonly_r3 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(CUDAToolkit REQUIRED)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/HBF-EventLab" eventlab)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper.ptx"
 COMMAND "${CMAKE_CUDA_COMPILER}" --ptx --relocatable-device-code=true -std=c++17 -arch=compute_75 "-I${ROOT}/include" "${ROOT}/src/async_ptx_helper.cu" -o "${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper.ptx"
 DEPENDS "${ROOT}/src/async_ptx_helper.cu" "${ROOT}/include/hbflab/async_device.cuh")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper_embedded.hpp"
 COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper.ptx" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper_embedded.hpp" -DSYMBOL=embedded_async_ptx_helper -P "${ROOT}/cmake/embed_ptx.cmake"
 DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper.ptx")
add_library(r3_runtime STATIC "${ROOT}/src/backend.cpp" "${ROOT}/src/async_runtime.cu" "${ROOT}/src/live_runtime.cpp" "${ROOT}/src/ptx.cpp" "${ROOT}/src/async_module.cpp" "${ROOT}/src/sha256.cpp" "${CMAKE_CURRENT_BINARY_DIR}/async_ptx_helper_embedded.hpp")
target_include_directories(r3_runtime PUBLIC "${ROOT}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(r3_runtime PUBLIC HBFEventLab::core CUDA::cudart CUDA::cuda_driver)
if(MSVC)
 target_compile_options(r3_runtime PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
 set_property(TARGET hbf_eventlab PROPERTY INTERFACE_COMPILE_OPTIONS "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
endif()
add_executable(r3_adaptive adaptive_gpu.cu)
target_include_directories(r3_adaptive PRIVATE "${ROOT}/include" "${ROOT}/third_party/HBF-EventLab/include")
target_link_libraries(r3_adaptive PRIVATE r3_runtime)
enable_testing()
foreach(case u8 u16 u32 u64 u128 partial sparse layout range drain)
 add_test(NAME adaptive_${case} COMMAND r3_adaptive ${case})
 set_tests_properties(adaptive_${case} PROPERTIES TIMEOUT 60)
endforeach()
add_executable(r3_run runner.cpp kernels.cu)
target_include_directories(r3_run PRIVATE "${ROOT}/third_party/llama.cpp-integrated/vendor")
target_link_libraries(r3_run PRIVATE r3_runtime)
target_compile_options(r3_run PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/EHsc>")
add_executable(r3_policy policy_test.cpp)
target_link_libraries(r3_policy PRIVATE r3_runtime)
add_test(NAME uniform_ptx_policy COMMAND r3_policy)
foreach(contract ptx_contract async_ptx_contract)
 add_executable(r3_${contract} "${ROOT}/tests/${contract}.cpp")
 target_link_libraries(r3_${contract} PRIVATE r3_runtime)
endforeach()
foreach(case hash coverage predicate reject_control reject_memory reject_module syntax multiple limits arity binary_comments access_manifest large)
 add_test(NAME ptx_${case} COMMAND r3_ptx_contract ${case})
endforeach()
foreach(case control labels predicate reject)
 add_test(NAME async_ptx_${case} COMMAND r3_async_ptx_contract ${case})
endforeach()
